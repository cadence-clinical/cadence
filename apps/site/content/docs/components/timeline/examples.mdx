## Filtered by kind

A [Toggle group](/docs/components/toggle-group) of kinds above the timeline. Pressing a kind shows only that kind. Pressing it again, or All, shows everything.

<TimelineFilterPreview />

```tsx
"use client";

import { ToggleGroup, ToggleGroupItem } from "@cadence-clinical/ui";
import { useState } from "react";

const FILTERS = [
  { value: "all", label: "All" },
  { value: "comments", label: "Comments" },
  { value: "notes", label: "Notes" },
] as const;

type Filter = (typeof FILTERS)[number]["value"];

export function Activity() {
  const [filter, setFilter] = useState<Filter>("all");
  return (
    <>
      <ToggleGroup
        aria-label="Show in the timeline"
        variant="outline"
        size="sm"
        spacing={0}
        value={[filter]}
        onValueChange={(value: string[]) => {
          // Nothing pressed means the pressed kind was released: show everything.
          setFilter(FILTERS.find((item) => item.value === value[0])?.value ?? "all");
        }}
      >
        {FILTERS.map((item) => (
          <ToggleGroupItem key={item.value} value={item.value}>
            {item.label}
          </ToggleGroupItem>
        ))}
      </ToggleGroup>
      {/* A Timeline of the items whose kind matches the filter */}
    </>
  );
}
```
